Описание процесса реализации Задания №1:
Для реализации поставленной задачи спроектирован внешний класс Payment, описывающий акт совершения покупки и содержащий атрибуты номера заказа (orderNumber), имени покупателя (customerName), даты оформления (date) и динамическую коллекцию товарных позиций (List<Item>). 
Внутри внешнего класса Payment объявлен внутренний (inner) нестатический класс Item (товарная позиция), представляющий конкретный приобретаемый товар. Внутренний класс инкапсулирует свойства: наименование товара (name), цену за единицу (price) и количество (quantity), а также вычисляет итоговую стоимость конкретной позиции методом getCost() (умножение цены на количество).
Внешний класс Payment предоставляет методы:
1. addItem(String name, double price, int quantity) — создание нового объекта внутреннего класса Item и добавление его в чек покупки;
2. calculateTotal() — итерационный подсчет суммарной стоимости всей покупки;
3. printReceipt() — форматированный вывод кассового чека с табличным перечислением позиций, цен, количества и итоговой суммы к оплате.

import java.util.ArrayList;
import java.util.List;

/**
 * Лабораторная работа №4 по дисциплине "Современные системы программирования"
 * Задание №1 (Приложение А, Вариант 2):
 * Создать класс Payment (покупка) с внутренним классом, с помощью объектов
 * которого можно сформировать покупку из нескольких товаров.
 */
public class Payment {
    private String orderNumber;
    private String customerName;
    private String date;
    private List<Item> items;

    /**
     * Внутренний (inner) класс, представляющий товарную позицию в чеке покупки.
     * Имеет прямой доступ к контексту внешней покупки.
     */
    public class Item {
        private String name;
        private double price;
        private int quantity;

        public Item(String name, double price, int quantity) {
            this.name = name;
            this.price = price;
            this.quantity = quantity;
        }

        public double getCost() {
            return price * quantity;
        }

        public String getName() {
            return name;
        }

        public double getPrice() {
            return price;
        }

        public int getQuantity() {
            return quantity;
        }

        @Override
        public String toString() {
            return String.format("%-26s %8.2f руб. x %2d шт. = %9.2f руб.", 
                    name, price, quantity, getCost());
        }
    }

    public Payment(String orderNumber, String customerName, String date) {
        this.orderNumber = orderNumber;
        this.customerName = customerName;
        this.date = date;
        this.items = new ArrayList<>();
    }

    // Добавление товара через создание объекта внутреннего класса
    public void addItem(String name, double price, int quantity) {
        Item newItem = new Item(name, price, quantity);
        items.add(newItem);
    }

    // Расчет общей суммы покупки
    public double calculateTotal() {
        double total = 0.0;
        for (Item item : items) {
            total += item.getCost();
        }
        return total;
    }

    // Печать детализированного кассового чека
    public void printReceipt() {
        System.out.println("==================================================================");
        System.out.println("                       КАССОВЫЙ ЧЕК ПОКУПКИ                       ");
        System.out.println("==================================================================");
        System.out.println("Номер заказа: " + orderNumber);
        System.out.println("Покупатель:   " + customerName);
        System.out.println("Дата покупки: " + date);
        System.out.println("------------------------------------------------------------------");
        System.out.println("№   Наименование товара            Цена       Кол-во    Стоимость");
        System.out.println("------------------------------------------------------------------");
        for (int i = 0; i < items.size(); i++) {
            System.out.printf("%2d. %s\n", (i + 1), items.get(i).toString());
        }
        System.out.println("------------------------------------------------------------------");
        System.out.printf("ИТОГО К ОПЛАТЕ:                                     %9.2f руб.\n", calculateTotal());
        System.out.println("==================================================================");
    }

    public static void main(String[] args) {
        System.out.println("Лабораторная работа №4. Задание №1 (Приложение А, Вариант 2)");
        System.out.println("Демонстрация работы класса Payment с внутренним классом Item:\n");

        Payment payment = new Payment("ЧЕК-2026/09-014", "Лабынцев Денис Янушевич", "16.09.2026 14:30");

        // Формирование покупки из нескольких товаров
        payment.addItem("Ноутбук ASUS Vivobook", 2450.00, 1);
        payment.addItem("Беспроводная мышь Logitech", 65.50, 2);
        payment.addItem("Механическая клавиатура", 180.00, 1);
        payment.addItem("Флеш-накопитель USB 64GB", 32.20, 3);
        payment.addItem("Коврик для мыши XL", 25.00, 1);

        // Вывод чека в консоль
        payment.printReceipt();
    }
}

// Дочерний класс 1: Коммерческая организация
class CommercialOrganization extends Organization {
    private double annualRevenue;  // Годовая выручка (руб.)
    private double profitTaxRate;   // Ставка налога на прибыль (%)
    private String directorName;    // Руководитель

    public CommercialOrganization(String name, String unp, String city, int foundationYear,
                                  double annualRevenue, double profitTaxRate, String directorName) {
        super(name, unp, city, foundationYear);
        this.annualRevenue = annualRevenue;
        this.profitTaxRate = profitTaxRate;
        this.directorName = directorName;
    }

    @Override
    public String getOrganizationType() {
        return "Коммерческая организация (ОАО / ООО)";
    }

    @Override
    public double calculateTax() {
        // Расчет налога на прибыль от годовой выручки (с учетом условной рентабельности)
        return annualRevenue * 0.15 * (profitTaxRate / 100.0);
    }

    @Override
    public void performActivity() {
        System.out.println("  Деятельность: производство промышленной продукции и оптово-розничная торговля");
    }

    @Override
    public void displayInfo() {
        super.displayInfo();
        System.out.println("  Руководитель:  " + directorName);
        System.out.printf("  Годовая выручка:  %,.2f руб. (Ставка налога: %.1f%%)\n", annualRevenue, profitTaxRate);
    }
}

Ответы на контрольные вопросы:

1. Что понимается под термином «наследование»?
Ответ: Наследование — это фундаментальный механизм объектно-ориентированного программирования, при котором характеристики и функциональность одного класса (суперкласса) передаются другому классу (подклассу) без необходимости их повторного описания, обеспечивая повторное использование кода и расширяемость.

2. Что общего имеет дочерний класс с родительским?
Ответ: Дочерний класс наследует структуру родителя: все его не-private поля, методы, конструктивные особенности и контракты интерфейсов, а также сохраняет отношение типизации «является» (is-a), что позволяет присваивать экземпляр потомка ссылке родительского типа.

3. В чем состоит различие между дочерним и родительским классами?
Ответ: Дочерний класс является более специализированным: он может дополнять родительскую структуру собственными новыми переменными и методами, а также переопределять (override) логику унаследованных методов для реализации полиморфизма.
