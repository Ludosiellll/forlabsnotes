Листинг 5 – Программная логика интерактивного слайдера (файл script.js)
const sliderTrack = document.getElementById("sliderTrack");
const slides = document.querySelectorAll(".slide");
const prevBtn = document.getElementById("prevBtn");
const nextBtn = document.getElementById("nextBtn");
const dots = document.querySelectorAll(".dot");

let currentSlide = 0;
const totalSlides = slides.length;
let autoSlideInterval = null;

function updateSlider(index) {
    currentSlide = (index + totalSlides) % totalSlides;
    const offset = -currentSlide * 100;
    if (sliderTrack) {
        sliderTrack.style.transform = `translateX(${offset}%)`;
    }

    dots.forEach((dot, idx) => {
        dot.classList.toggle("active", idx === currentSlide);
    });
}

if (nextBtn) {
    nextBtn.addEventListener("click", () => {
        updateSlider(currentSlide + 1);
        resetAutoplay();
    });
}

if (prevBtn) {
    prevBtn.addEventListener("click", () => {
        updateSlider(currentSlide - 1);
        resetAutoplay();
    });
}

dots.forEach((dot) => {
    dot.addEventListener("click", (e) => {
        const index = parseInt(e.target.getAttribute("data-index"), 10);
        updateSlider(index);
        resetAutoplay();
    });
});

function startAutoplay() {
    autoSlideInterval = setInterval(() => {
        updateSlider(currentSlide + 1);
    }, 4500);
}

function resetAutoplay() {
    if (autoSlideInterval) {
        clearInterval(autoSlideInterval);
        startAutoplay();
    }
}

startAutoplay();

Контрольные вопросы

1. Hover-эффекты и их применение?
Ответ: Hover-эффекты представляют собой визуальные изменения элементов (вращение, сдвиг, трансформация, смена цвета), активируемые псевдоклассом :hover при наведении курсора мыши. Они применяются для визуального отклика интерфейса, акцентирования внимания пользователя на кликабельных компонентах и создания динамичного вида веб-сайта.

2. Назовите наиболее распространенные hover-эффекты?
Ответ: К наиболее распространенным относятся изменение цвета текста или фона, трансформация рамки, масштабирование (увеличение или сжатие), эффект перекатывания (rollover), смещение фона через background-position, появление иконок и трехмерный переворот.

3. Алгоритм действия эффекта перекатывания?
Ответ: Алгоритм включает подготовку двух изображений одинакового размера, установку первого изображения фоном блочного элемента <a>, а второго — фоном при псевдоклассе :hover либо вертикальное смещение спрайта с помощью свойства background-position: 0 -Ypx.

4. Применение hover-эффектов к кнопкам?
Ответ: К кнопкам применяются горизонтальное и вертикальное заполнение полупрозрачным фоном с переходом transition: all 500ms ease, плавное выдвижение иконки слева от текста, пружинящее подпрыгивание @keyframes bounce и объемный 3D-переворот с псевдоэлементом :after и transform-style: preserve-3d.

Вывод: Сформировал умение создавать визуальные эффекты и анимации на HTML – странице.
